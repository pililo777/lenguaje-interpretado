Sistema de Gráficos

← Volver al índice

1. Creación de Ventanas

// Crear ventana
ventana nombre ancho alto

// Mostrar ventana
mostrar ventana nombre

// Cambiar título
cambiar_titulo ventana nombre "nuevo título"

2. Componentes Gráficos

// Botones
botón ventana nombre "texto" x y

// Etiquetas
etiqueta ventana nombre "texto" x y

// Texto editable
texto ventana nombre x y

3. Dibujo en Pantalla

// Líneas
dibujar línea x1 y1 x2 y2

// Circulos
dibujar círculo x y radio

// Rectángulos
dibujar rectángulo x1 y1 x2 y2

// Puntos
dibujar punto x y

4. Manejo de Eventos

// Eventos de ventana
ventana nombre "evento" procedimiento

// Eventos de botón
botón nombre "evento" procedimiento

// Eventos de texto
texto nombre "evento" procedimiento

5. Ejemplo Completo

// Crear ventana principal
ventana principal 400 300

// Agregar componentes
botón principal "saludar" "Saludar" 50 50
etiqueta principal "mensaje" "" 50 100

// Definir procedimiento
procedimiento saludar
    mensaje = "¡Hola!"
fin

// Mostrar ventana
mostrar ventana principal
