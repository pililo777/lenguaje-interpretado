1. Lexer (tokens.l)

// Tokens principales
"="     { return EQ; }
"<>"    { return NE; }
"<"     { return LT; }
">"     { return GT; }
"+"     { return PLUS; }
"-"     { return MINUS; }
"*"     { return MULT; }
"/"     { return DIVIDE; }

// Palabras clave
"if"|"IF"|"SI"|"si"   { return IF; }
"while"|"WHILE"|"MIENTRAS"|"mientras" { return WHILE; }
"for"|"FOR"|"DESDE"|"desde" { return FOR; }
"print"|"PRINT"|"IMPRIMIR"|"imprimir"|"?"  { return PRINT; }
"stop" { return STOP; }
"abrir" { return ABRIR; }
"cerrar" { return CERRAR; }
"mostrar" { return MOSTRAR; }
"vaciar" { return VACIAR; }
"continuar" { return CONTINUAR; }
"salir" { return SALIR; }

// Literales
'\'(\.|[^'])*\'  { return LITERAL; }
"\"(\.|[^\"])*\"  { return STRING; }

2. Gramática (grammar.y)

%token BUSCAR INSERTAR ELIMINAR USE_INDICE CLOSE_INDICE
%token STOP  REGISTRO FINREGISTRO

CONVERTIR EVALUAR

// Reglas principales
ROOT:
    stmtseq    {  pila_programas[idx_prg] = ($1); idx_prg++; }
|  stmtseq TERMINAR procedimientos { pila_programas[idx_prg] = ($1); idx_prg++; }
;

// Declaraciones
statement:
    designator EQ expression { $$ = nodo2(asigna_num, $1, $3); }
| sdesignator EQ STRING { $$ = nodo2(asigna_alfa, $1, $3); }
| DIM designator NUMBER { $$ = nodo2(dimensionar, $2, $3); }
| designator '[' expression ']' EQ expression { $$ = nodo3(asigna_vector, $1, $3, $6); }

// Control de flujo
| IF expression THEN stmtseq ELSE stmtseq FI { $$ = nodo3(si, $2, $4, $6); }
| WHILE expression DO stmtseq OD { $$ = nodo2(mientras, $2, $4); }
| FOR designator EQ expression TO expression DO stmtseq OD { $$ = nodo4(desde, $2, $4, $6, $8); }

// Salida
| PRINT lista_expr { $$ = nodo1(imprimir_varios, $2); }
| MENSAJE lista_expr { $$ = nodo1(mensaje, $2); }

3. Características Principales

  • Soporte para variables numéricas y alfanuméricas
  • Manejo de vectores y matrices
  • Operaciones aritméticas y lógicas
  • Control de flujo (if, while, for)
  • Manejo de archivos
  • Interfaz gráfica (GTK+)
  • Sistema de índices
